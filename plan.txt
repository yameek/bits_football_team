Technical Specifications


CREATE TABLE members (
    id INT PRIMARY KEY,
    name VARCHAR(100),
    contact_number VARCHAR(20),
    balance DECIMAL(10,2)
);

CREATE TABLE fields (
    id INT PRIMARY KEY,
    name VARCHAR(100),
    address TEXT,
    google_map_link VARCHAR(255)
);

CREATE TABLE sessions (
    id INT PRIMARY KEY,
    field_id INT,
    scheduled_date DATETIME,
    session_type ENUM('practice', 'match'),
    field_cost DECIMAL(10,2),
    transport_cost DECIMAL(10,2),
    drinks_cost DECIMAL(10,2),
    emergency_fund DECIMAL(10,2),
    FOREIGN KEY (field_id) REFERENCES fields(id)
);

CREATE TABLE attendance (
    id INT PRIMARY KEY,
    member_id INT,
    session_id INT,
    present BOOLEAN,
    FOREIGN KEY (member_id) REFERENCES members(id),
    FOREIGN KEY (session_id) REFERENCES sessions(id)
);

CREATE TABLE transactions (
    id INT PRIMARY KEY,
    member_id INT,
    transaction_type ENUM('contribution', 'field_fee', 'transport', 'drinks'),
    amount DECIMAL(10,2),
    timestamp DATETIME,
    FOREIGN KEY (member_id) REFERENCES members(id)
);
